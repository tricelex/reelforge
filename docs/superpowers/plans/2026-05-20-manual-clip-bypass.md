# Manual Clip Bypass Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Allow operators to bypass AI analysis when uploading pre-edited videos, fast-forwarding straight to the candidate configure stage with a single full-video manual clip candidate.

**Architecture:** Add `skip_analysis` to `ClippingJob` and `is_manual` to `ClipCandidate`. A new `skip_to_candidates` FSM transition + `_create_manual_candidate` helper drive all three entry points: the post-transcription task branch, a new API action, and a new admin action. Candidate duration/overlap validation is bypassed when `is_manual=True`.

**Tech Stack:** Django 5.2, django-fsm-2, Celery, DRF, drf-spectacular, Unfold admin, ffmpeg-python, pytest + factory_boy.

def _create_manual_candidate(job: ClippingJob) -> ClipCandidate:
    """Create a full-video manual ClipCandidate for a job that bypassed analysis.

    Probes the downloaded file for duration when source_duration_sec is unset
    (UPLOAD source type does not populate it during download).
    """
    duration = job.source_duration_sec
    if duration is None and job.downloaded_file:
        from django.conf import settings as _settings

        probe = ffmpeg.probe(str(Path(_settings.MEDIA_ROOT) / job.downloaded_file.name))
        duration = float(probe["format"]["duration"])
        job.source_duration_sec = duration
        job.save(update_fields=["source_duration_sec", "updated_at"])
    return ClipCandidate.objects.create(

        start_sec=0.0,

        is_manual=True,

    @admin.action(description="Fast-forward to candidates (skip AI analysis)")
    def fast_forward_to_candidates(
        self, request: HttpRequest, queryset: QuerySet[ClippingJob]
    ) -> None:
        from ***REMOVED***.clipping.tasks import _create_manual_candidate
        from ***REMOVED***.clipping.sse import emit_job_event

        forwarded = 0
        skipped = 0
        for job in queryset:
            if not can_proceed(job.skip_to_candidates):
                skipped += 1
                continue
            try:
                job.skip_to_candidates()
                job.save(update_fields=["status", "updated_at"])
                _create_manual_candidate(job)
                emit_job_event(str(job.id), "status_changed", {"status": job.status})
                forwarded += 1
            except Exception as exc:

        if forwarded:

                request,
                f"Fast-forwarded {forwarded} job(s) to candidate stage.",
                messages.SUCCESS,
            )
        if skipped:
            self.message_user(

```python
    @extend_schema(
        tags=["clipping-jobs"],
        summary="Fast-forward a job to candidate stage, skipping AI analysis",
        description=(
            "Creates a single is_manual=True ClipCandidate spanning the full video "
            "and transitions the job to AWAITING_CLIP_APPROVAL. "
            "Valid from TRANSCRIBING, ANALYZING, or FAILED states."
        ),
        request=None,
        responses={
            200: ClippingJobDetailSerializer,
            400: OpenApiResponse(description="Job cannot be fast-forwarded from its current state"),
        },
    )
    @action(detail=True, methods=["post"], url_path="skip-to-candidates")
    def skip_to_candidates(self, request: Request, pk: str | None = None) -> Response:
        job: ClippingJob = self.get_object()
        if not can_proceed(job.skip_to_candidates):
            return Response(
                {"detail": f"Cannot skip to candidates from state {job.status}."},
                status=status.HTTP_400_BAD_REQUEST,
            )
        job.skip_to_candidates()
        job.save(update_fields=["status", "updated_at"])
        _create_manual_candidate(job)
        emit_job_event(str(job.id), "status_changed", {"status": job.status})
        logger.info("Job fast-forwarded to candidates via API", extra={"job_id": str(job.id)})
        return Response(
            ClippingJobDetailSerializer(job, context=self.get_serializer_context()).data
        )
```

- No request body
- Returns: full job object (same shape as `GET /api/v1/clipping/jobs/{id}/`)

async function handleSkipToCandiates(jobId: string) {

    const res = await fetch(`/api/v1/clipping/jobs/${jobId}/skip-to-candidates/`, {
