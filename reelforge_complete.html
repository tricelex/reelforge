class YouTubeClientError(Exception):
    """Base exception for all YouTube client errors."""


class YouTubeAuthError(YouTubeClientError):
    """Raised when OAuth fails, token is invalid, or scope is insufficient."""


class YouTubeQuotaError(YouTubeClientError):
    """Raised when the YouTube API quota is exceeded."""


class YouTubeAPIError(YouTubeClientError):
    """Raised for general non-auth, non-quota API errors."""

    def __init__(self, message: str, status_code: int | None = None):
        self.status_code = status_code
        super().__init__(message)

import httpx
from datetime import timedelta
from urllib.parse import urlencode
from django.conf import settings
from django.utils import timezone
from .exceptions import YouTubeAuthError


SCOPES = [
    "https://www.googleapis.com/auth/youtube.readonly",
    "https://www.googleapis.com/auth/youtube.upload",
    "https://www.googleapis.com/auth/yt-analytics.readonly",
]

GOOGLE_TOKEN_URL = "https://oauth2.googleapis.com/token"
GOOGLE_AUTH_URL = "https://accounts.google.com/o/oauth2/v2/auth"


def get_authorization_url(redirect_uri: str, state: str | None = None) -> str:
    """Generate the Google OAuth2 authorization URL to redirect users to."""
    params = {
        "client_id": settings.GOOGLE_CLIENT_ID,
        "redirect_uri": redirect_uri,
        "response_type": "code",
        "scope": " ".join(SCOPES),
        "access_type": "offline",
        "prompt": "consent",
    }
    if state:
        params["state"] = state
    return f"{GOOGLE_AUTH_URL}?{urlencode(params)}"


def exchange_code_for_tokens(code: str, redirect_uri: str) -> dict:
    """Exchange an OAuth2 authorization code for access and refresh tokens."""
    response = httpx.post(GOOGLE_TOKEN_URL, data={
        "code": code,
        "client_id": settings.GOOGLE_CLIENT_ID,
        "client_secret": settings.GOOGLE_CLIENT_SECRET,
        "redirect_uri": redirect_uri,
        "grant_type": "authorization_code",
    })
    if response.status_code != 200:
        raise YouTubeAuthError(f"Token exchange failed: {response.text}")
    return response.json()


def refresh_access_token(refresh_token: str) -> dict:
    """Use a refresh token to obtain a new access token."""
    response = httpx.post(GOOGLE_TOKEN_URL, data={
        "client_id": settings.GOOGLE_CLIENT_ID,
        "client_secret": settings.GOOGLE_CLIENT_SECRET,
        "refresh_token": refresh_token,
        "grant_type": "refresh_token",
    })
    if response.status_code != 200:
        raise YouTubeAuthError(f"Token refresh failed: {response.text}")
    return response.json()

import httpx
from datetime import timedelta
from django.conf import settings
from django.utils import timezone
from .exceptions import YouTubeAPIError, YouTubeAuthError, YouTubeQuotaError
from .oauth import refresh_access_token

BASE_URL = "https://www.googleapis.com/youtube/v3"


class YouTubeClient:
    """
    General-purpose YouTube Data API v3 client.

    Two operating modes:
      - Public  (API key):  search, trending, video/channel stats — no user auth needed.
      - OAuth   (token):    channel management, uploads, analytics — requires user consent.

    Usage:
        # Public data
        client = YouTubeClient()

        # Authenticated (pass stored credential)
        client = YouTubeClient.from_credential(credential)
    """

    def __init__(
        self,
        api_key: str | None = None,
        access_token: str | None = None,
        credential=None,
    ):
        self.api_key = api_key or settings.YOUTUBE_API_KEY
        self.access_token = access_token
        self._credential = credential
        self.http = httpx.Client(timeout=15)

    @classmethod
    def from_credential(cls, credential) -> "YouTubeClient":
        """Instantiate an authenticated client from a YouTubeCredential model instance."""
        instance = cls(credential=credential)
        instance._ensure_fresh_token()
        return instance

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------

    def _ensure_fresh_token(self):
        if not self._credential:
            return
        if self._credential.is_expired:
            token_data = refresh_access_token(self._credential.refresh_token)
            self._credential.access_token = token_data["access_token"]
            self._credential.token_expiry = timezone.now() + timedelta(
                seconds=token_data.get("expires_in", 3600)
            )
            self._credential.save(update_fields=["access_token", "token_expiry"])
        self.access_token = self._credential.access_token

    def _get_headers(self) -> dict:
        if self.access_token:
            return {"Authorization": f"Bearer {self.access_token}"}
        return {}

    def _build_params(self, params: dict) -> dict:
        if not self.access_token:
            params["key"] = self.api_key
        return params

    def _get(self, endpoint: str, params: dict) -> dict:
        response = self.http.get(
            f"{BASE_URL}/{endpoint}",
            params=self._build_params(params),
            headers=self._get_headers(),
        )
        self._handle_errors(response)
        return response.json()

    def _handle_errors(self, response: httpx.Response):
        if response.status_code == 200:
            return
        if response.status_code == 401:
            raise YouTubeAuthError("Invalid or expired credentials.")
        if response.status_code == 403:
            errors = response.json().get("error", {}).get("errors", [])
            if any(e.get("reason") == "quotaExceeded" for e in errors):
                raise YouTubeQuotaError("YouTube API quota exceeded.")
            raise YouTubeAuthError("Access forbidden. Check OAuth scopes.")
        raise YouTubeAPIError(
            f"YouTube API error: {response.text}",
            status_code=response.status_code,
        )

    def _get_video_stats(self, video_ids: list[str]) -> dict:
        if not video_ids:
            return {}
        data = self._get("videos", {"part": "statistics", "id": ",".join(video_ids)})
        return {i["id"]: i["statistics"] for i in data.get("items", [])}

    def _require_auth(self):
        if not self.access_token:
            raise YouTubeAuthError("This method requires OAuth authentication.")

    # ------------------------------------------------------------------
    # Public data — API key sufficient
    # ------------------------------------------------------------------

    def search_videos(
        self,
        query: str,
        max_results: int = 10,
        order: str = "viewCount",
    ) -> list[dict]:
        """Search YouTube videos. Enriches results with statistics."""
        data = self._get("search", {
            "part": "snippet",
            "q": query,
            "maxResults": max_results,
            "type": "video",
            "order": order,
        })
        items = data.get("items", [])
        video_ids = [i["id"]["videoId"] for i in items]
        stats = self._get_video_stats(video_ids)
        for item in items:
            item["statistics"] = stats.get(item["id"]["videoId"], {})
        return items

    def get_trending_videos(
        self,
        region_code: str = "US",
        category_id: str = "0",
        max_results: int = 10,
    ) -> list[dict]:
        """Fetch trending videos by region and category."""
        data = self._get("videos", {
            "part": "snippet,statistics",
            "chart": "mostPopular",
            "regionCode": region_code,
            "videoCategoryId": category_id,
            "maxResults": max_results,
        })
        return data.get("items", [])

    def get_video_details(self, video_id: str) -> dict:
        """Get full details for a single video."""
        data = self._get("videos", {
            "part": "snippet,statistics,contentDetails",
            "id": video_id,
        })
        items = data.get("items", [])
        return items[0] if items else {}

    def get_channel_videos(
        self,
        channel_id: str,
        max_results: int = 10,
        order: str = "viewCount",
    ) -> list[dict]:
        """Fetch videos from a channel ordered by a given criterion."""
        data = self._get("search", {
            "part": "snippet",
            "channelId": channel_id,
            "maxResults": max_results,
            "order": order,
            "type": "video",
        })
        return data.get("items", [])

    def get_channel_stats(self, channel_id: str) -> dict:
        """Get stats for any public YouTube channel by ID. No OAuth required."""
        data = self._get("channels", {
            "part": "snippet,statistics",
            "id": channel_id,
        })
        items = data.get("items", [])
        if not items:
            raise YouTubeAPIError(f"Channel {channel_id} not found.")
        item = items[0]
        stats = item.get("statistics", {})
        snippet = item.get("snippet", {})
        return {
            "channel_id": item["id"],
            "title": snippet.get("title"),
            "description": snippet.get("description"),
            "subscriber_count": int(stats.get("subscriberCount", 0) or 0),
            "view_count": int(stats.get("viewCount", 0) or 0),
            "video_count": int(stats.get("videoCount", 0) or 0),
            "published_at": snippet.get("publishedAt"),
        }

    def get_video_categories(self, region_code: str = "US") -> list[dict]:
        """List available video categories for a region."""
        data = self._get("videoCategories", {
            "part": "snippet",
            "regionCode": region_code,
        })
        return data.get("items", [])

    # ------------------------------------------------------------------
    # Authenticated operations — OAuth required
    # ------------------------------------------------------------------

    def get_my_channel(self) -> dict:
        """Get full details for the authenticated user's channel."""
        self._require_auth()
        data = self._get("channels", {
            "part": "snippet,statistics,brandingSettings,contentDetails",
            "mine": True,
        })
        items = data.get("items", [])
        return items[0] if items else {}

    def get_my_channel_stats(self) -> dict:
        """Get a clean summary of the authenticated user's channel statistics."""
        self._require_auth()
        channel = self.get_my_channel()
        stats = channel.get("statistics", {})
        snippet = channel.get("snippet", {})
        return {
            "channel_id": channel.get("id"),
            "title": snippet.get("title"),
            "description": snippet.get("description"),
            "subscriber_count": stats.get("subscriberCount"),
            "view_count": stats.get("viewCount"),
            "video_count": stats.get("videoCount"),
            "published_at": snippet.get("publishedAt"),
        }

    def get_my_videos(self, max_results: int = 25) -> list[dict]:
        """List uploaded videos for the authenticated user's channel."""
        self._require_auth()
        channel = self.get_my_channel()
        uploads_playlist_id = (
            channel.get("contentDetails", {})
            .get("relatedPlaylists", {})
            .get("uploads")
        )
        if not uploads_playlist_id:
            return []
        data = self._get("playlistItems", {
            "part": "snippet,contentDetails",
            "playlistId": uploads_playlist_id,
            "maxResults": max_results,
        })
        return data.get("items", [])

    def get_my_playlists(self, max_results: int = 25) -> list[dict]:
        """List playlists belonging to the authenticated user."""
        self._require_auth()
        data = self._get("playlists", {
            "part": "snippet,contentDetails,status",
            "mine": True,
            "maxResults": max_results,
        })
        return data.get("items", [])

    def get_my_comments(self, video_id: str, max_results: int = 20) -> list[dict]:
        """Get top-level comments on one of the authenticated user's videos."""
        self._require_auth()
        data = self._get("commentThreads", {
            "part": "snippet",
            "videoId": video_id,
            "maxResults": max_results,
            "order": "relevance",
        })
        return data.get("items", [])

from ...services.youtube.client import YouTubeClient

class YouTubeProvider:
    """
    Satisfies the VideoSearchProvider protocol.

    Delegates all HTTP to YouTubeClient (services layer) and normalizes
    raw API responses into VideoResult dataclasses for agent consumption.
    """

    def __init__(self, client: YouTubeClient | None = None):
        self._client = client or YouTubeClient()

    def search_videos(self, query: str, max_results: int = 10) -> list[VideoResult]:
        raw = self._client.search_videos(query, max_results)
        return [self._normalize(item) for item in raw]

    def get_trending(
        self, region_code: str = "US", category_id: str = "0"
    ) -> list[VideoResult]:
        raw = self._client.get_trending_videos(region_code, category_id)
        return [self._normalize(item, use_id_direct=True) for item in raw]

    def get_channel_videos(
        self, channel_id: str, max_results: int = 10
    ) -> list[VideoResult]:
        raw = self._client.get_channel_videos(channel_id, max_results)
        return [self._normalize(item) for item in raw]

    def _normalize(self, item: dict, use_id_direct: bool = False) -> VideoResult:
        video_id = item["id"] if use_id_direct else item["id"]["videoId"]
        stats = item.get("statistics", {})
        return VideoResult(
            title=item["snippet"]["title"],
            url=f"https://youtube.com/watch?v={video_id}",
            channel=item["snippet"]["channelTitle"],
            views=int(stats.get("viewCount", 0) or 0),
            likes=int(stats.get("likeCount", 0) or 0),
            published_at=item["snippet"]["publishedAt"],
        )

import httpx
from django.conf import settings
from ..schemas import VideoResult, SearchReport, SearchResult, TrendData


class SerpApiProvider:
    """
    Satisfies the WebSearchProvider protocol.
    Handles Google Search, YouTube SERP, and Google Trends via SerpApi.
    """

    BASE_URL = "https://serpapi.com/search"

    def __init__(self, api_key: str | None = None):
        self.api_key = api_key or settings.SERPAPI_KEY
        self.http = httpx.Client(timeout=15)

    def _get(self, params: dict) -> dict:
        response = self.http.get(
            self.BASE_URL,
            params={**params, "api_key": self.api_key},
        )
        response.raise_for_status()
        return response.json()

    def search(self, query: str) -> SearchReport:
        """Google search — returns organic results, PAA, and related searches."""
        raw = self._get({"engine": "google", "q": query})
        return SearchReport(
            organic_results=[
                SearchResult(
                    title=r["title"],
                    url=r["link"],
                    snippet=r.get("snippet", ""),
                )
                for r in raw.get("organic_results", [])[:5]
            ],
            people_also_ask=[
                q["question"] for q in raw.get("related_questions", [])
            ],
            related_searches=[
                r.get("query", "") for r in raw.get("related_searches", [])
            ],
        )

    def search_youtube(self, query: str) -> list[VideoResult]:
        """YouTube SERP search via SerpApi."""
        raw = self._get({"engine": "youtube", "search_query": query})
        return [
            VideoResult(
                title=v["title"],
                url=v.get("link", ""),
                channel=v.get("channel", {}).get("name", ""),
                views=v.get("views"),
                likes=None,
                published_at=v.get("published_date"),
            )
            for v in raw.get("video_results", [])[:8]
        ]

    def get_trends(self, keyword: str, region: str = "US") -> TrendData:
        """Google Trends data for a keyword."""
        raw = self._get({
            "engine": "google_trends",
            "q": keyword,
            "geo": region,
        })
        return TrendData(
            keyword=keyword,
            interest_over_time=raw.get("interest_over_time", {}).get("timeline_data", []),
            related_queries=[
                q["query"]
                for q in raw.get("related_queries", {}).get("rising", [])[:10]
            ],
        )

class AgentContainer(containers.DeclarativeContainer):
    """
    DI container for all agent dependencies.

    To swap a provider, override the singleton here — nothing else changes:
        web_search = providers.Singleton(BraveSearchProvider, api_key=config.brave_api_key)

    To override in tests:
        with AgentContainer.video_search.override(mock_provider):
            ...
    """

    config = providers.Configuration()

    # Core service clients
    youtube_client = providers.Singleton(
        YouTubeClient,
        api_key=config.youtube_api_key,
    )

    # Providers — these are what tools consume
    video_search = providers.Singleton(
        YouTubeProvider,
        client=youtube_client,
    )

    web_search = providers.Singleton(
        SerpApiProvider,
        api_key=config.serpapi_key,
    )

@function_tool

def get_google_trends(

PydanticField — schema-enforced JSONField that returns Pydantic model instances
# ***REMOVED***/fields.py

from typing import Type, TypeVar
from django.db import models
from pydantic import BaseModel

T = TypeVar("T", bound=BaseModel)


class PydanticField(models.JSONField):
    """
    A JSONField that validates against a Pydantic model schema.

    - Reads from DB  → returns a Pydantic model instance
    - Writes to DB   → serializes Pydantic instance to JSON automatically
    - ORM queries    → still work (it's a JSONField under the hood)

    Usage:
        research_report = PydanticField(schema=ResearchReport)
    """

    def __init__(self, schema: Type[T], *args, **kwargs):
        self.schema = schema
        kwargs.setdefault("null", True)
        kwargs.setdefault("blank", True)
        super().__init__(*args, **kwargs)

    def deconstruct(self):
        name, path, args, kwargs = super().deconstruct()
        kwargs["schema"] = self.schema
        return name, path, args, kwargs

    def from_db_value(self, value, expression, connection):
        value = super().from_db_value(value, expression, connection)
        if value is None:
            return None
        if isinstance(value, dict):
            return self.schema.model_validate(value)
        return value

    def to_python(self, value):
        value = super().to_python(value)
        if value is None:
            return None
        if isinstance(value, dict):
            return self.schema.model_validate(value)
        if isinstance(value, self.schema):
            return value
        return value

    def get_prep_value(self, value):
        if isinstance(value, BaseModel):
            value = value.model_dump()
        return super().get_prep_value(value)

    ]

    @admin.action(description="↺ Sync public channel stats from YouTube")
    def sync_channel_stats(self, request, queryset):
        # Uses API key — no OAuth required for public channel stats
        client = YouTubeClient()
        success, failed = 0, 0

        for channel in queryset:
            try:
                data = client.get_channel_stats(channel.channel_id)
                channel.title = data["title"]
                channel.description = data.get("description", "")
                channel.subscriber_count = data["subscriber_count"]
                channel.view_count = data["view_count"]
                channel.video_count = data["video_count"]
                channel.stats_last_synced = timezone.now()
                channel.save(update_fields=[
                    "title", "description", "subscriber_count",
                    "view_count", "video_count", "stats_last_synced",
                ])
                success += 1

            except YouTubeQuotaError:
                self.message_user(request, "YouTube API quota exceeded. Try again later.", messages.ERROR)
                return

            except (YouTubeAuthError, Exception) as e:
                self.message_user(
                    request,
                    f"Failed to sync {channel.channel_id}: {e}",
                    messages.ERROR,
                )
                failed += 1

        if success:
            self.message_user(request, f"Synced {success} channel(s).", messages.SUCCESS)
        if failed:
            self.message_user(request, f"{failed} channel(s) failed.", messages.WARNING)

    @admin.action(description="↺ Sync authenticated user's own channel stats")
    def sync_my_channel_stats(self, request, queryset):
        try:
            credential = request.user.youtube_credential
        except YouTubeCredential.DoesNotExist:
            self.message_user(request, "No YouTube account connected for your user.", messages.ERROR)
            return

        try:
            client = YouTubeClient.from_credential(credential)
            stats = client.get_my_channel_stats()
            # Update whichever channel in the selection matches
            queryset.filter(channel_id=stats["channel_id"]).update(
                title=stats["title"],
                subscriber_count=stats["subscriber_count"],
                view_count=stats["view_count"],
                video_count=stats["video_count"],
                stats_last_synced=timezone.now(),
            )
            self.message_user(request, f"Synced: {stats['title']}", messages.SUCCESS)
        except Exception as e:
            self.message_user(request, f"Failed: {e}", messages.ERROR)

from celery import shared_task
from django.utils import timezone
from agents import Runner
from .models import ResearchJob
from .agents.research_agent import build_research_agent
from .agents.schemas import ResearchReport


@shared_task(bind=True, max_retries=2)
def run_research_job(self, job_id: int):
    """
    Run the research agent for a given ResearchJob.
    Called from the Django admin action or programmatically via .delay(job_id).
    """
    job = ResearchJob.objects.get(id=job_id)
    job.status = ResearchJob.Status.RUNNING
    job.save(update_fields=["status"])

    try:
        agent = build_research_agent()

        prompt = f"Research this topic for a YouTube video: {job.topic}"
        if job.niche:
            prompt += f"\nNiche context: {job.niche}"
        if job.target_region:
            prompt += f"\nTarget region: {job.target_region}"

        result = Runner.run_sync(agent, prompt, max_turns=15)

        # Validate agent output against Pydantic schema — raises if malformed
        report = ResearchReport.model_validate_json(result.final_output)

        job.status = ResearchJob.Status.COMPLETED
        job.research_report = report  # assign Pydantic instance — PydanticField handles serialization
        job.completed_at = timezone.now()
        job.error_message = ""
        job.save(update_fields=["status", "research_report", "completed_at", "error_message"])

    except Exception as exc:
        job.status = ResearchJob.Status.FAILED
        job.error_message = str(exc)
        job.save(update_fields=["status", "error_message"])
        raise self.retry(exc=exc, countdown=60)
